// Get trial key const r = await fetch("https://api.anchorwithin.com/v1/trial/request", { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ email: "you@yourapp.com" }) }); const { api_key } = await r.json(); // Stream audio to file (Node.js) const resp = await fetch("https://api.anchorwithin.com/v1/audio/serve", { method: "POST", headers: { "Authorization": `Bearer ${api_key}`, "Content-Type": "application/json" }, body: JSON.stringify({ frequency: "528", use_case: "healing" }) }); const buffer = await resp.arrayBuffer(); require("fs").writeFileSync("528hz.mp3", Buffer.from(buffer));
